Lumina Grammar (EBNF)

          
Formal language grammar & operator reference.

        

        
      

      

        


<Program>          ::= <StatementList> <EOF>
<StatementList>    ::= <Statement>*

<Statement>        ::= <VarDeclStmt>
                     | <AssignStmt>
                     | <PrintStmt>
                     | <IfStmt>
                     | <WhileStmt>
                     | <HaltStmt>
                     | <BlockStmt>

<VarDeclStmt>      ::= ("Bindint" | "Bindstrng" | "Bindbool") <ID> ":=" <Expr>
<AssignStmt>       ::= <ID> ":=" <Expr>
<PrintStmt>        ::= "Turnout" <Expr>
<IfStmt>           ::= "examine" <Expr> <Statement> [ "otherwise" <Statement> ]
<WhileStmt>        ::= "noose" <Expr> <Statement>
<BlockStmt>        ::= "{" <StatementList> "}"
<HaltStmt>         ::= "halt"

<Expr>             ::= <Relational>
<Relational>       ::= <Additive> [ ( ">>" | "<<" | ">=;" ) <Additive> ]
<Additive>         ::= <Multiplicative> ( ( "++>" | "<--" ) <Multiplicative> )*
<Multiplicative>   ::= <Primary> ( ( "***" | "://" ) <Primary> )*
<Primary>          ::= <NUMBER> | <STRING> | "True" | "False" | <ID> | "(" <Expr> ")"
